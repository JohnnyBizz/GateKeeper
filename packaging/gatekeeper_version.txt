# Windows version resource for GateKeeper.exe.
#
# This is not decoration. A PyInstaller executable with no version resource is
# a nameless binary that nothing has ever seen before, and that is most of
# what a browser's download check and Defender's heuristics have to go on: no
# publisher, no product, no version, self-extracting stub. Filling it in does
# not make the file signed, but it moves it out of the shape that gets
# refused on sight.
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(1, 0, 0, 0),
    prodvers=(1, 0, 0, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0),
  ),
  kids=[
    StringFileInfo([
      StringTable(
        '040904B0',
        [StringStruct('CompanyName', 'GateKeeper'),
         StringStruct('FileDescription',
                      'GateKeeper - technical analysis assistant'),
         StringStruct('FileVersion', '1.0.0.0'),
         StringStruct('InternalName', 'GateKeeper'),
         StringStruct('LegalCopyright', 'Open source. No warranty.'),
         StringStruct('OriginalFilename', 'GateKeeper.exe'),
         StringStruct('ProductName', 'GateKeeper'),
         StringStruct('ProductVersion', '1.0.0.0')])
    ]),
    VarFileInfo([VarStruct('Translation', [1033, 1200])])
  ]
)
